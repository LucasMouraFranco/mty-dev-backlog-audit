(function () {
  const q = document.getElementById('q');
  const chips = Array.from(document.querySelectorAll('.chip'));
  const cards = Array.from(document.querySelectorAll('.issue-card'));
  const needsAttention = document.getElementById('needsAttention');
  const allIssues = document.getElementById('allIssues');
  const emptyState = document.getElementById('emptyState');
  const resultCount = document.getElementById('resultCount');
  const restTotal = document.getElementById('restTotal');
  let filter = 'all';

  function matchesFilter(card) {
    const state = card.dataset.state;
    const match = card.dataset.match;
    if (filter === 'all') return true;
    if (filter === 'todo') return state === 'todo';
    if (filter === 'backlog') return state === 'backlog';
    if (filter === 'strong') return match === 'strong';
    if (filter === 'weak') return match === 'weak';
    if (filter === 'none') return match === 'none';
    return true;
  }

  function matchesSearch(card) {
    const term = q.value.trim().toLowerCase();
    if (!term) return true;
    return (card.dataset.search || '').includes(term);
  }

  function apply() {
    let visible = 0;
    let strongVisible = 0;
    let restVisible = 0;
    cards.forEach(card => {
      const ok = matchesFilter(card) && matchesSearch(card);
      card.classList.toggle('hidden', !ok);
      if (ok) {
        visible++;
        if (card.dataset.match === 'strong' && card.closest('#strongList')) strongVisible++;
        if (card.closest('#restList')) restVisible++;
      }
    });
    // Show/hide sections
    const showStrongSection = filter === 'all' || filter === 'strong' ||
      (['todo','backlog'].includes(filter) && strongVisible > 0) ||
      (q.value.trim() !== '' && strongVisible > 0);
    // For weak/none filters, hide needs-attention section
    if (filter === 'weak' || filter === 'none') {
      needsAttention.classList.add('hidden');
    } else if (filter === 'strong') {
      needsAttention.classList.remove('hidden');
      allIssues.classList.add('hidden');
    } else if (filter === 'todo' || filter === 'backlog') {
      needsAttention.classList.toggle('hidden', strongVisible === 0);
      allIssues.classList.toggle('hidden', restVisible === 0 && strongVisible > 0 && false);
      allIssues.classList.remove('hidden');
      if (restVisible === 0 && strongVisible === 0) allIssues.classList.add('hidden');
    } else {
      // all
      needsAttention.classList.remove('hidden');
      allIssues.classList.remove('hidden');
    }
    if (filter === 'strong') {
      // already handled
    } else if (filter === 'weak' || filter === 'none') {
      allIssues.classList.remove('hidden');
    }

    // When searching, hide empty sections
    if (q.value.trim()) {
      needsAttention.classList.toggle('hidden', strongVisible === 0);
      if (filter !== 'strong') {
        allIssues.classList.toggle('hidden', restVisible === 0);
      }
    }

    emptyState.classList.toggle('hidden', visible !== 0);
    if (visible === 0) {
      allIssues.classList.remove('hidden');
      needsAttention.classList.add('hidden');
    }

    const label = q.value.trim() || filter !== 'all'
      ? `Showing ${visible} of 90 issues`
      : `90 issues`;
    resultCount.textContent = label;
    if (restTotal) restTotal.textContent = String(restVisible || 86);
    // Keep static count label for section when all+no search
    if (!q.value.trim() && filter === 'all') {
      restTotal.textContent = '86';
    }
  }

  chips.forEach(chip => {
    chip.addEventListener('click', () => {
      chips.forEach(c => c.classList.remove('active'));
      chip.classList.add('active');
      filter = chip.dataset.filter;
      apply();
    });
  });
  q.addEventListener('input', apply);
  apply();
})();
</script>
